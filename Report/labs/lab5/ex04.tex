\task{For the voltage-divider bias circuit below, find $V_G$, $I_{DQ}$, $V_{GSQ}$, $V_D$, $V_S$ and $V_{DS}$.
Verify the operating point with LTspice.}

\subsubsection*{Theoretical Calculation}
\begin{enumerate}[itemsep=2pt]
  \item Compute $V_G$ from the divider (gate current is zero).
  \item Write $V_{GS} = V_G - I_D R_S$, combine with Shockley's equation and solve for $I_D$.
  \item Compute $V_S$, $V_D$, $V_{DS}$ and check the region of operation.
\end{enumerate}
\answerspace{10}
\begin{solution}
\[
  V_G = V_{DD}\frac{R_2}{R_1+R_2} = 12\cdot\frac{2.2}{5.5} = 4.8\,\mathrm{V},\qquad V_{GS} = 4.8 - 3.3\,I_D \;(I_D\ \text{in mA})
\]
\[
  I_D = 4\left(1+\frac{4.8-3.3I_D}{2}\right)^2 = 4(3.4-1.65I_D)^2 \;\Rightarrow\; 10.89I_D^2 - 45.88I_D + 46.24 = 0
  \;\Rightarrow\; I_D = 1.669\ \text{or}\ 2.544\ \mathrm{mA}
\]
$I_D = 2.544$\,mA gives $V_{GS} = -3.60\,\mathrm{V} < V_P$ (not valid), so $I_{DQ} = 1.67$\,mA and $V_{GSQ} = -0.71$\,V.
\[
  V_S = 1.669\cdot3.3 = 5.51\,\mathrm{V},\qquad V_D = 12 - 1.669\cdot 1.8 = 9.00\,\mathrm{V},\qquad
  V_{DS} = 3.49\,\mathrm{V} > V_{GS}-V_P = 1.29\,\mathrm{V}
\]
The JFET operates in the saturation region.
\end{solution}
