\task{For the voltage-divider bias circuit below, find $V_G$, $I_{DQ}$, $V_{GSQ}$, $V_D$, $V_S$ and $V_{DS}$.
Verify the operating point with LTspice.}

\subsubsection*{Circuit}
\circuitfig{ex04}{Circuit of Exercise 5.4: JFET voltage-divider bias}
\components{
\begin{itemize}[leftmargin=*,itemsep=1pt]
  \item $V_{DD} = 16$\,V, $R_1 = 3\,\mathrm{M}\Omega$ (top), $R_2 = 1\,\mathrm{M}\Omega$ (bottom), $R_D = 2.2\,\mathrm{k}\Omega$, $R_S = 2\,\mathrm{k}\Omega$.
  \item J1: \texttt{.model JN NJF(Vto=-2 Beta=1m Lambda=0)}.
\end{itemize}}

\subsubsection*{Theoretical Calculation}
\begin{enumerate}[itemsep=2pt]
  \item Compute $V_G$ from the divider (gate current is zero).
  \item Write $V_{GS} = V_G - I_D R_S$, combine with Shockley's equation and solve for $I_D$.
  \item Compute $V_S$, $V_D$, $V_{DS}$ and check the region of operation.
\end{enumerate}
\answerspace{10}
\begin{solution}
\[
  V_G = V_{DD}\frac{R_2}{R_1+R_2} = 16\cdot\frac{1}{4} = 4\,\mathrm{V},\qquad V_{GS} = 4 - 2I_D \;(I_D\ \text{in mA})
\]
\[
  I_D = 4\left(1+\frac{4-2I_D}{2}\right)^2 = 4(3-I_D)^2 \;\Rightarrow\; 4I_D^2 - 25I_D + 36 = 0 \;\Rightarrow\; I_D = 2.25\ \text{or}\ 4\ \mathrm{mA}
\]
$I_D = 4$\,mA gives $V_{GS} = -4\,\mathrm{V} < V_P$ (not valid), so $I_{DQ} = 2.25$\,mA and $V_{GSQ} = -0.5$\,V.
\[
  V_S = 4.5\,\mathrm{V},\qquad V_D = 16 - 2.25\cdot 2.2 = 11.05\,\mathrm{V},\qquad V_{DS} = 6.55\,\mathrm{V} > V_{GS}-V_P = 1.5\,\mathrm{V}
\]
\end{solution}
