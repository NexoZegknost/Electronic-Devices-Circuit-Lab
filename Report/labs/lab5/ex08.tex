\exercise{MOSFET as a Switch}{Keep}{Ch.6 4.1 -- MOSFET as a switch}
\statusreason{Same circuit, values and objective as the original ($R_{IN} = 4.7\,\mathrm{k}\Omega$, $R_{GS} = 47\,\mathrm{k}\Omega$,
TTL input, $100\,\Omega$ load instead of the lamp, $V_{DD} = 12$\,V). A switching-type level-1 \texttt{.model} replaces the
PSpice part so that $R_{DS(\mathrm{on})}$ can be estimated by hand.}

\task{An n-channel E-MOSFET switches a $100\,\Omega$ load (replacing a lamp) from $V_{DD} = 12$\,V. The gate is driven by a
TTL signal (0\,V / 5\,V) through $R_{IN}$, and $R_{GS}$ pulls the gate down when the input is low.
Determine $V_{GS}$, the load current, $V_{DS}$ and the load power in the ON state, and $V_{DS}$ in the OFF state.
Verify with a transient simulation.}

\subsubsection*{Circuit}
\circuitfig{ex08}{Circuit of Exercise 5.8: MOSFET as a switch}
\components{
\begin{itemize}[leftmargin=*,itemsep=1pt]
  \item $V_{DD} = 12$\,V, $R_L = 100\,\Omega$, $R_{IN} = 4.7\,\mathrm{k}\Omega$, $R_{GS} = 47\,\mathrm{k}\Omega$.
  \item $V_{IN}$: \texttt{PULSE(0 5 0 10n 10n 0.5m 1m)} (1\,kHz, 50\,\% duty cycle).
  \item M1 (symbol \texttt{nmos4}, $L = W = 1\,\mu$m): \texttt{.model SW NMOS(Level=1 Vto=2 Kp=0.5 Lambda=0)}
\end{itemize}}

\subsubsection*{Theoretical Calculation}
\begin{enumerate}[itemsep=2pt]
  \item Compute $V_{GS}$ in the ON state from the divider $R_{IN}$--$R_{GS}$.
  \item In the ohmic region with small $V_{DS}$, $R_{DS(\mathrm{on})} \approx 1/[K_P(V_{GS}-V_{GS(\mathrm{Th})})]$. Compute it.
  \item Compute the load current $I_L$, $V_{DS(\mathrm{on})}$ and the load power $P_L$. What is $V_{DS}$ when the input is low?
\end{enumerate}
\answerspace{10}
\begin{solution}
\[
  V_{GS} = 5\cdot\frac{47}{4.7+47} = 4.55\,\mathrm{V},\qquad
  R_{DS(\mathrm{on})} \approx \frac{1}{0.5\cdot(4.55-2)} = 0.79\,\Omega
\]
\[
  I_L = \frac{12}{100+0.79} = 119\,\mathrm{mA},\qquad V_{DS(\mathrm{on})} = I_L R_{DS(\mathrm{on})} \approx 0.094\,\mathrm{V},\qquad
  P_L = I_L^2 R_L \approx 1.42\,\mathrm{W}
\]
Solving the full triode equation $I_D = K_P[(V_{GS}-V_{GS(\mathrm{Th})})V_{DS} - V_{DS}^2/2]$ gives $V_{DS} = 0.095$\,V,
practically the same. In the OFF state $V_{GS} = 0$, $I_D = 0$ and $V_{DS} = V_{DD} = 12$\,V.
\end{solution}

\subsubsection*{Simulation}
\begin{enumerate}[itemsep=2pt]
  \item Run the transient analysis below. Plot \texttt{V(in)}, \texttt{V(g)}, \texttt{V(d)} and \texttt{I(RL)}.
  \item Read $V_{GS}$, $I_L$ and $V_{DS}$ in the ON half-period and $V_{DS}$ in the OFF half-period.
\end{enumerate}

\begin{solution}
Agreement is expected; the level-1 model has no gate capacitance by default, so the edges are as fast as the input.
The ON current (about 119\,mA, just below the 120\,mA grid line) and the OFF voltage of 12\,V agree with the theory;
$V_{GS}$ is read as 4.5\,V on the grid.
