\subsubsection*{Theoretical Calculation}
\begin{enumerate}[itemsep=2pt]
  \item For a level-1 model, $I_D = \tfrac{K_P}{2}\tfrac{W}{L}(V_{GS}-V_{to})^2$ in saturation. Show that this is Shockley's
        equation with $V_P = V_{to}$ and $I_{DSS} = \tfrac{K_P}{2}V_P^2$, and compute $I_{DSS}$.
  \item Compute $I_D$ at $V_{GS} = -2$, $-1$ and $+1$\,V.
  \item (b) Solve the self-bias circuit for $I_{DQ}$, $V_{GSQ}$ and $V_{DS}$.
\end{enumerate}
\answerspace{10}
\begin{solution}
With $W/L = 1$: $I_D = 1\,\mathrm{mA/V^2}\,(V_{GS}+3)^2 = 9\,\mathrm{mA}\,(1+V_{GS}/3)^2$, so $V_P = -3$\,V and $I_{DSS} = 9$\,mA.
$I_D(-2\,\mathrm{V}) = 1$\,mA, $I_D(-1\,\mathrm{V}) = 4$\,mA, $I_D(+1\,\mathrm{V}) = 16$\,mA (enhancement region).

In part (a), at $V_{GS} = +1$\,V the drain voltage is $V_{DS} = 10 - 0.1\cdot16 = 8.4$\,V, still in saturation.

(b) $V_{GS} = -I_D R_4 = -I_D$ ($I_D$ in mA, $R_4 = 1\,\mathrm{k}\Omega$):
\[
  I_D = 9\left(1-\frac{I_D}{3}\right)^2 \;\Rightarrow\; I_D^2 - 7I_D + 9 = 0 \;\Rightarrow\; I_D = 1.697\ \text{or}\ 5.303\ \mathrm{mA}
\]
$I_D = 5.303$\,mA gives $V_{GS} = -5.30\,\mathrm{V} < V_P$ (not valid), so $I_{DQ} = 1.70$\,mA, $V_{GSQ} = -1.70$\,V,
$V_D = 12 - 1.697\cdot2 = 8.61$\,V, $V_S = 1.70$\,V, $V_{DS} = 6.91\,\mathrm{V} > V_{GS}-V_P = 1.30$\,V (saturation).
\end{solution}
