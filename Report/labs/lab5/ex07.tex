\task{In the drain-feedback bias circuit below, the gate is connected to the drain through a large resistor $R_G$.
Determine $I_{DQ}$, $V_{GSQ}$ and $V_{DSQ}$, and verify them with an operating-point analysis.}

\subsubsection*{Circuit}
\circuitfig{ex07}{Circuit of Exercise 5.7: E-MOSFET drain-feedback bias}
\components{
\begin{itemize}[leftmargin=*,itemsep=1pt]
  \item $V_{DD} = 10$\,V, $R_D = 3\,\mathrm{k}\Omega$, $R_G = 10\,\mathrm{M}\Omega$ (drain to gate), source and bulk grounded.
  \item M1: \texttt{.model EM NMOS(Level=1 Vto=2 Kp=1m Lambda=0)} \quad ($V_{GS(\mathrm{Th})} = 2$\,V, $k = 0.5\,\mathrm{mA/V^2}$).
\end{itemize}}

\subsubsection*{Theoretical Calculation}
\begin{enumerate}[itemsep=2pt]
  \item Explain why $V_{GS} = V_{DS}$ in this circuit.
  \item Write $V_{GS} = V_{DD} - I_D R_D$, combine with $I_D = k(V_{GS}-V_{GS(\mathrm{Th})})^2$ and solve for $V_{GS}$.
  \item Compute $I_{DQ}$ and check the region of operation.
\end{enumerate}
\answerspace{10}
\begin{solution}
$I_G = 0$, so no voltage drops across $R_G$ and $V_G = V_D$, i.e.\ $V_{GS} = V_{DS}$.
\[
  \frac{10 - V_{GS}}{3\,\mathrm{k}\Omega} = 0.5\,\mathrm{mA/V^2}\,(V_{GS}-2)^2
  \;\Rightarrow\; 1.5V_{GS}^2 - 5V_{GS} - 4 = 0 \;\Rightarrow\; V_{GS} = 4\ \text{or}\ -0.67\ \mathrm{V}
\]
The negative root is below the threshold (not valid), so $V_{GSQ} = V_{DSQ} = 4$\,V and $I_{DQ} = (10-4)/3 = 2$\,mA.
Since $V_{DS} = V_{GS} > V_{GS}-V_{GS(\mathrm{Th})}$, the device is always in saturation in this configuration.
\end{solution}
