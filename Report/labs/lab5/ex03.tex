\task{For the self-bias circuit below, determine the quiescent values $I_{DQ}$, $V_{GSQ}$, $V_D$, $V_S$ and $V_{DS}$.
Check that the JFET operates in the saturation region, then verify the results with an operating-point analysis.}

\subsubsection*{Circuit}
\circuitfig{ex03}{Circuit of Exercise 5.3: JFET self-bias}
\components{
\begin{itemize}[leftmargin=*,itemsep=1pt]
  \item $V_{DD} = 12$\,V, $R_D = 2\,\mathrm{k}\Omega$, $R_S = 1\,\mathrm{k}\Omega$, $R_G = 680\,\mathrm{k}\Omega$ (gate to ground).
  \item J1: \texttt{.model JN NJF(Vto=-2 Beta=1m Lambda=0)} \quad ($I_{DSS} = 4$\,mA, $V_P = -2$\,V).

\subsubsection*{Theoretical Calculation}
\begin{enumerate}[itemsep=2pt]
  \item Write the self-bias line $V_{GS} = -I_D R_S$ (no gate current, $V_G = 0$).
  \item Substitute it into Shockley's equation and solve the quadratic equation for $I_D$. Reject the root that gives $V_{GS} < V_P$.
  \item Compute $V_S$, $V_D$ and $V_{DS}$, and check $V_{DS} > V_{GS} - V_P$.
\end{enumerate}
\answerspace{10}
\begin{solution}
With $I_D$ in mA and $R_S = 1\,\mathrm{k}\Omega$: $V_{GS} = -I_D$ (V).
\[
  I_D = 4\left(1 - \frac{I_D}{2}\right)^2 \;\Rightarrow\; I_D^2 - 5I_D + 4 = 0 \;\Rightarrow\; I_D = 1\ \text{or}\ 4\ \mathrm{mA}
\]
$I_D = 4$\,mA gives $V_{GS} = -4\,\mathrm{V} < V_P$ (not valid), so
$I_{DQ} = 1$\,mA, $V_{GSQ} = -1$\,V.
\[
  V_S = I_D R_S = 1\,\mathrm{V},\qquad V_D = V_{DD} - I_D R_D = 10\,\mathrm{V},\qquad V_{DS} = V_D - V_S = 9\,\mathrm{V}
\]
$V_{DS} = 9\,\mathrm{V} > V_{GS}-V_P = 1\,\mathrm{V}$: saturation region.
\end{solution}
